            ( Value /text= CStr
                { csValue = "# Bind
                The bind operator [<-] is syntactic sugar that runs the rest of a function in a sub-context.
                A statement [x <- m] followed by the remaining body [rest] desugars to
                [bindap(m= m, fn(v)= rest)], where the continuation [fn] binds [x] (the desugar inlines [rest] as
                a higher-order lambda, so a bind-using function needs a declared return type to pin the result).
                [bindap] has overloaded arms selected by the value's type and the continuation: [mmap] for a
                [Monad] and [fmap] for a [Functor]. A value that is both (e.g. [Maybe]) may take either arm —
                the arms are alternatives ([#exist]), and the result type the call must reach chooses: a
                continuation returning a [Maybe] into a declared [Maybe[$T -> Integer]] can only be the
                [mmap] arm, since [fmap] would nest the result. The [ap] arm (a plain value) is kept below,
                commented out, until its dispatch is implemented." }
            )
        , /Catln/#md
            ( Value /text= CStr
                { csValue = "[mmap] arm: a [Monad]. The continuation maps an element to a monad of the same kind; the result" }
            )
        , /Catln/#md
            ( Value /text= CStr
                { csValue = "is flattened back into one monad (the monadic bind)." }

            )

            ( Value /text= CStr
                { csValue = "container of the mapped elements." }
            )
        ]
