                    ( Value lines= /Data/List/Cons
                        ( Value /head= CStr

                        ( Value /tail= /Data/List/Cons
                            ( Value /head= CStr
                                { csValue = "b" }
                            )
                            ( Value /tail= /Data/List/Cons
                                ( Value /head= CStr
                                    { csValue = "a" }
                                )( Value /tail= Value /Data/List/Nil )
                            )
                        )
                    )( Value acc= Value /Data/List/Nil )
                )
            )
            ( Value /result= CStr
                { csValue = "a,b,c" }
            )

                The parts of a module, each written as the text it is in a `.ll` file." }

                    )( Value /tail= Value /Data/List/Nil )
                )( Value externs= Value /Data/List/Nil )

                define i32 @main() {

                  ret i32 0
